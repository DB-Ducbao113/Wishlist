@echo off
:: ============================================================
:: DESIGN-01/02 Visual Evidence Capture Script
:: Personal Vault V16.2 — Design Reconstruction
:: ============================================================

:: REQUIREMENTS:
:: - Node.js installed (>=14.0.0)
:: - Playwright CLI installed: npm i -D playwright
:: - The app built: npm run build
:: - App served locally: npx serve dist -l 3000

:: STEP 1: Install Playwright if not already installed
if not exist "%APPDATA%\npm\playwright.cmd" (
    echo Installing Playwright...
    npm init -y >nul 2>&1
    npm i -D playwright >nul 2>&1
    npx playwright install >nul 2>&1
)

:: STEP 2: Capture desktop 1440x900 screenshots
echo.
echo Capturing desktop 1440x900 screenshots...
npx playwright test --project=chromium --viewport-size=1440x900 --base-url=http://localhost:3000 --test-name-pattern="visual capture" >nul 2>&1

:: Create and run the capture script
cat > capture-visual.js << 'ENDSCRIPT'
const { chromium } = require('playwright');

(async () => {
  const browser = await chromium.launch({ headless: false });
  const context = await browser.newContext({
    viewport: { width: 1440, height: 900 }
  });
  const page = await context.newPage();
  
  // Navigate to the Personal Vault app
  await page.goto('http://localhost:3000');
  await page.waitForLoadState('networkidle');
  
  // Wait for hero and dashboard to render
  await page.waitForSelector('.hero, .dashboard', { timeout: 10000 });
  
  // 1. Full page screenshot
  await page.screenshot({ path: 'design-01-02-reference-1440-full.png', fullPage: true });
  console.log('Captured: desktop 1440x900 full page');
  
  // 2. Hero section screenshot
  await page.screenshot({ path: 'design-01-02-reference-1440-hero.png', clip: { x: 0, y: 0, width: 1440, height: 500 } });
  console.log('Captured: desktop 1440x900 hero section');
  
  // 3. Dashboard section screenshot  
  await page.screenshot({ path: 'design-01-02-reference-1440-dashboard.png', clip: { x: 0, y: 700, width: 1440, height: 500 } });
  console.log('Captured: desktop 1440x900 dashboard section');
  
  // 4. Navigation at top (scrollY=0)
  await page.evaluate(() => { window.scrollTo(0, 0); });
  await page.waitForTimeout(500);
  await page.screenshot({ path: 'design-01-02-reference-1440-nav-top.png', clip: { x: 0, y: 0, width: 1440, height: 80 } });
  console.log('Captured: desktop 1440x900 navigation at top');
  
  // 5. Navigation at middle position
  await page.evaluate(() => { window.scrollTo(0, 700); });
  await page.waitForTimeout(500);
  await page.screenshot({ path: 'design-01-02-reference-1440-nav-middle.png', clip: { x: 0, y: 0, width: 1440, height: 80 } });
  console.log('Captured: desktop 1440x900 navigation at middle');
  
  // 6. Navigation at bottom
  await page.evaluate(() => { window.scrollTo(0, document.body.scrollHeight); });
  await page.waitForTimeout(500);
  await page.screenshot({ path: 'design-01-02-reference-1440-nav-bottom.png', clip: { x: 0, y: 0, width: 1440, height: 80 } });
  console.log('Captured: desktop 1440x900 navigation at bottom');
  
  // 7. Mobile 390x844
  const context2 = await browser.newContext({ viewport: { width: 390, height: 844 } });
  const page2 = await context2.newPage();
  await page2.goto('http://localhost:3000');
  await page2.waitForLoadState('networkidle');
  await page2.waitForSelector('.hero, .dashboard', { timeout: 10000 });
  
  await page2.screenshot({ path: 'design-01-02-reference-390-full.png', fullPage: true });
  console.log('Captured: mobile 390x844 full page');
  
  // 8. Mobile navigation at top
  await page2.evaluate(() => { window.scrollTo(0, 0); });
  await page2.waitForTimeout(500);
  await page2.screenshot({ path: 'design-01-02-reference-390-nav-top.png', clip: { x: 0, y: 0, width: 390, height: 80 } });
  console.log('Captured: mobile 390x844 navigation at top');
  
  // 9. Mobile navigation at middle
  await page2.evaluate(() => { window.scrollTo(0, 500); });
  await page2.waitForTimeout(500);
  await page2.screenshot({ path: 'design-01-02-reference-390-nav-middle.png', clip: { x: 0, y: 0, width: 390, height: 80 } });
  console.log('Captured: mobile 390x844 navigation at middle');
  
  // 10. Mobile navigation at bottom
  await page2.evaluate(() => { window.scrollTo(0, document.body.scrollHeight); });
  await page2.waitForTimeout(500);
  await page2.screenshot({ path: 'design-01-02-reference-390-nav-bottom.png', clip: { x: 0, y: 0, width: 390, height: 80 } });
  console.log('Captured: mobile 390x844 navigation at bottom');
  
  await browser.close();
  console.log('\nAll screenshots captured successfully!');
  console.log('Compare these with: references/someday-vault-motion-reference.html');
})();
ENDSCRIPT
